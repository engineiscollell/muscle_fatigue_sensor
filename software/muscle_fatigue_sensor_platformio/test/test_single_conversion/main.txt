#include <Arduino.h>
#include <SPI.h>

#include "config.h"

// ============================================================================
// REGISTRES AD7172
// ============================================================================

static constexpr uint8_t REG_STATUS      = 0x00;
static constexpr uint8_t REG_ADCMODE     = 0x01;
static constexpr uint8_t REG_IFMODE      = 0x02;
static constexpr uint8_t REG_DATA        = 0x04;
static constexpr uint8_t REG_ID          = 0x07;

static constexpr uint8_t REG_CH0         = 0x10;

static constexpr uint8_t REG_SETUPCON0   = 0x20;
static constexpr uint8_t REG_FILTCON0    = 0x28;

// ============================================================================
// COMMUNICATION REGISTER
// ============================================================================

static constexpr uint8_t COMM_READ = 0x40;

// ============================================================================
// SPI SETTINGS
// ============================================================================

SPISettings spiSettings(
    1000000,
    MSBFIRST,
    SPI_MODE3
);

// ============================================================================
// RESET ADC
// ============================================================================

void adc_reset()
{
    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    // 64 clocks amb DIN=1
    for (int i = 0; i < 8; i++)
    {
        SPI.transfer(0xFF);
    }

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();

    delay(5);
}

// ============================================================================
// READ ID REGISTER (16 BIT)
// ============================================================================

uint16_t adc_read_id()
{
    uint16_t value = 0;

    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    SPI.transfer(COMM_READ | REG_ID);

    value |= ((uint16_t)SPI.transfer(0x00) << 8);
    value |= ((uint16_t)SPI.transfer(0x00));

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();

    return value;
}

// ============================================================================
// WRITE REGISTER 16 BIT
// ============================================================================

void adc_write_register16(uint8_t reg, uint16_t value)
{
    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    // bit6 = 0 -> WRITE
    SPI.transfer(reg);

    SPI.transfer((value >> 8) & 0xFF);
    SPI.transfer(value & 0xFF);

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();
}

// ============================================================================
// READ REGISTER 16 BIT
// ============================================================================

uint16_t adc_read_register16(uint8_t reg)
{
    uint16_t value = 0;

    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    // bit6 = 1 -> READ
    SPI.transfer(COMM_READ | reg);

    value |= ((uint16_t)SPI.transfer(0x00) << 8);
    value |= ((uint16_t)SPI.transfer(0x00));

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();

    return value;
}

// ============================================================================
// READ DATA REGISTER (24 BIT)
// ============================================================================

uint32_t adc_read_data24()
{
    uint32_t value = 0;

    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    SPI.transfer(COMM_READ | REG_DATA);

    value |= ((uint32_t)SPI.transfer(0x00) << 16);
    value |= ((uint32_t)SPI.transfer(0x00) << 8);
    value |= ((uint32_t)SPI.transfer(0x00));

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();

    return value;
}

// ============================================================================
// START SINGLE CONVERSION
// ============================================================================

void adc_start_single_conversion()
{
    SPI.beginTransaction(spiSettings);

    digitalWrite(PIN_ADC_CS, LOW);

    SPI.transfer(REG_ADCMODE);

    // REF_EN + SINGLE CONVERSION
    SPI.transfer(0x80);
    SPI.transfer(0x10);

    digitalWrite(PIN_ADC_CS, HIGH);

    SPI.endTransaction();
}

// ============================================================================
// SETUP
// ============================================================================

void setup()
{
    Serial.begin(115200);

    delay(1000);

    Serial.println();
    Serial.println("=== AD7172 CONVERSION TEST ===");

    // =========================================================================
    // GPIO
    // =========================================================================

    pinMode(PIN_ADC_CS, OUTPUT);
    digitalWrite(PIN_ADC_CS, HIGH);

    pinMode(PIN_ADC_DRDY, INPUT_PULLUP);

    // =========================================================================
    // SPI
    // =========================================================================

    SPI.begin();

    // =========================================================================
    // RESET ADC
    // =========================================================================

    adc_reset();

    Serial.printf(
        "DRDY state: %d\n",
        digitalRead(PIN_ADC_DRDY)
    );

    // =========================================================================
    // READ ID
    // =========================================================================

    uint16_t id = adc_read_id();

    Serial.printf(
        "ADC ID: 0x%04X\n",
        id
    );

    // =========================================================================
    // CONFIGURACIÓ ADC
    // =========================================================================

    // IFMODE
    adc_write_register16(REG_IFMODE, 0x0100);

    // SETUP0
    adc_write_register16(REG_SETUPCON0, 0x0020);

    // FILT0
    adc_write_register16(REG_FILTCON0, 0x000B);

    // CHANNEL0
    // Enable + AIN0/AIN1
    adc_write_register16(REG_CH0, 0x8001);

    Serial.println("ADC configurat");

    uint16_t setup =
        adc_read_register16(REG_SETUPCON0);

    uint16_t filt =
        adc_read_register16(REG_FILTCON0);

    uint16_t ch0 =
        adc_read_register16(REG_CH0);

    Serial.printf("SETUP0 : 0x%04X\n", setup);
    Serial.printf("FILT0  : 0x%04X\n", filt);
    Serial.printf("CH0    : 0x%04X\n", ch0);

    // =========================================================================
    // READBACK TEST
    // =========================================================================

    uint16_t ifmode =
        adc_read_register16(REG_IFMODE);

    Serial.printf(
        "IFMODE: 0x%04X\n",
        ifmode
    );

    // =========================================================================
    // START SINGLE CONVERSION
    // =========================================================================

    adc_start_single_conversion();

    uint16_t adcmode =
        adc_read_register16(REG_ADCMODE);

    Serial.printf(
        "ADCMODE: 0x%04X\n",
        adcmode
    );

    Serial.println("Conversio iniciada");

    // =========================================================================
    // WAIT DRDY
    // =========================================================================

    uint32_t timeout = millis();

    while (digitalRead(PIN_ADC_DRDY))
    {
        if ((millis() - timeout) > 100)
        {
            Serial.println("Timeout DRDY");
            return;
        }
    }

    Serial.println("DRDY LOW detectat");

    // =========================================================================
    // READ SAMPLE
    // =========================================================================

    uint32_t sample = adc_read_data24();

    Serial.printf(
        "SAMPLE: %lu (0x%06lX)\n",
        sample,
        sample
    );

    uint8_t status =
        adc_read_register16(REG_STATUS) & 0xFF;

    Serial.printf(
        "STATUS: 0x%02X\n",
        status
    );
}

// ============================================================================
// LOOP
// ============================================================================

void loop()
{
}
